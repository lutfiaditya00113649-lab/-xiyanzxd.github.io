let curService='all', curGame='all', cart=[];
const listEl=document.getElementById('list');

function formatRp(n){return 'Rp '+n.toLocaleString('id-ID');}

function render(){
  listEl.innerHTML='';
  produk.filter(p=>(curService==='all'||p.service===curService)&&(curGame==='all'||p.game===curGame)).forEach(p=>{

function addToCart(id){
  const item=produk.find(p=>p.id===id);
  cart.push(item);
  updateCart();
  openCart();
}

function updateCart(){
  document.getElementById('cart-count').innerText=cart.length;
  document.getElementById('cart-count2').innerText=cart.length;
  const wrap=document.getElementById('cart-items');
  let total=0;
  wrap.innerHTML='';
  cart.forEach((c,i)=>{
    total+=c.price;
    wrap.innerHTML+=`<div class="bg-[#121827] p-3 rounded-xl flex justify-between items-center"><div><p class="text-xs font-bold">${c.title}</p><p class="text-xs text-blue-400">${formatRp(c.price)}</p></div><button onclick="removeCart(${i})" class="text-red-400"><i class="fas fa-trash"></i></button></div>`;
  });
  document.getElementById('cart-total').innerText=formatRp(total);
}

function removeCart(i){cart.splice(i,1); updateCart();}
function clearCart(){cart=[]; updateCart();}
function openCart(){document.getElementById('cart').classList.remove('hidden');}
function closeCart(){document.getElementById('cart').classList.add('hidden');}

function checkoutWA(){
  if(cart.length===0) return alert('Keranjang kosong!');
  const nama=document.getElementById('nama').value||'-';
  const uid=document.getElementById('uid').value||'-';
  let pesan=`Halo Xiyanz X'D mau checkout:%0A%0A`;
  let total=0;
  cart.forEach((c,i)=>{pesan+=`${i+1}. ${c.title} - ${formatRp(c.price)}%0A`; total+=c.price;});
  pesan+=`%0ATotal: ${formatRp(total)}%0ANama: ${nama}%0AUID/ID: ${uid}`;
  window.open(`https://wa.me/${WA}?text=${pesan}`,'_blank');
}
